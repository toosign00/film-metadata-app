- [ ] **Step 2: Header 컴포넌트 전체 교체**

`src/components/layout/Header/index.tsx`를 다음으로 교체한다:

```tsx
'use client';

import { Check, Moon, RotateCcw, Sun } from 'lucide-react';
import { useTheme } from 'next-themes';
import { useEffect, useState } from 'react';
import type { HeaderProps } from '@/types/header.types';

const steps = [
  { number: 1, label: '파일 선택' },
  { number: 2, label: '메타데이터 설정' },
  { number: 3, label: '결과 확인' },
];

export const Header = ({
  className,
  activeStep,
  filesCount = 0,
  resultsCount = 0,
  processing = false,
  goToStep,
  resetForm,
}: HeaderProps) => {
  const [mounted, setMounted] = useState(false);
  const [confirmReset, setConfirmReset] = useState(false);
  const { resolvedTheme, setTheme } = useTheme();

  useEffect(() => {
    setMounted(true);
  }, []);

  const isStepEnabled = (stepNumber: number) => {
    if (processing) return false;
    if (stepNumber === 1) return true;
    if (stepNumber === 2) return filesCount > 0;
    if (stepNumber === 3) return resultsCount > 0;
    return false;
  };

  const isStepCompleted = (stepNumber: number) => {
    if (activeStep === undefined) return false;
    return stepNumber < activeStep;
  };

  const handleResetClick = () => {
    if (confirmReset) {
      resetForm?.();
      setConfirmReset(false);
    } else {
      setConfirmReset(true);
      setTimeout(() => setConfirmReset(false), 3000);
    }
  };

  const showStepper = activeStep !== undefined && goToStep !== undefined;

  return (
    <header className={`border-b border-border bg-surface ${className || ''}`}>
      <div className='mx-auto max-w-6xl px-6 py-6 md:px-8'>
        <div className='flex items-start justify-between'>
          <div>
            <h1 className='text-[22px] font-bold tracking-[-0.6px] text-foreground leading-tight'>
              필름 사진 메타데이터 설정 도구
            </h1>
            <p className='mt-1 text-sm text-foreground-muted'>
              스캔된 필름 사진에 EXIF 메타데이터를 일괄 설정합니다
            </p>
          </div>
          <div className='ml-4 flex shrink-0 items-center gap-2 mt-1'>
            {showStepper && resetForm && (
              <button
                type='button'
                onClick={handleResetClick}
                disabled={processing || filesCount === 0}
                className='cursor-pointer rounded-md p-1.5 text-foreground-muted transition-colors hover:bg-surface-alt hover:text-destructive disabled:cursor-not-allowed disabled:opacity-40'
                aria-label={confirmReset ? '한 번 더 클릭하면 초기화됩니다' : '모두 초기화'}
                title={confirmReset ? '한 번 더 클릭하면 초기화됩니다' : '모두 초기화'}
              >
                <RotateCcw size={16} className={confirmReset ? 'text-destructive' : ''} />
              </button>
            )}
            {mounted ? (
              <button
                type='button'
                onClick={() => setTheme(resolvedTheme === 'dark' ? 'light' : 'dark')}
                className='cursor-pointer rounded-md border border-border p-1.5 text-foreground-muted transition-colors hover:bg-surface-alt hover:text-foreground'
                aria-label={resolvedTheme === 'dark' ? '라이트 모드로 전환' : '다크 모드로 전환'}
              >
                {resolvedTheme === 'dark' ? <Sun size={16} /> : <Moon size={16} />}
              </button>
            ) : (
              <div className='h-[30px] w-[30px] shrink-0' />
            )}
          </div>
        </div>

        {showStepper && (
          <div className='mt-6 flex items-center'>
            {steps.map((step, index) => {
              const enabled = isStepEnabled(step.number);
              const completed = isStepCompleted(step.number);
              const active = activeStep === step.number;

              return (
                <div key={step.number} className='flex items-center'>
                  <button
                    type='button'
                    onClick={() => enabled && goToStep(step.number)}
                    disabled={!enabled}
                    className={`flex items-center gap-2 ${enabled ? 'cursor-pointer' : 'cursor-not-allowed'}`}
                    aria-current={active ? 'step' : undefined}
                  >
                    <div
                      className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold transition-colors ${
                        completed
                          ? 'bg-primary text-primary-foreground'
                          : active
                            ? 'bg-foreground text-surface'
                            : 'border border-border-hover text-foreground-muted'
                      }`}
                    >
                      {completed ? <Check size={12} strokeWidth={3} /> : step.number}
                    </div>
                    <span
                      className={`hidden text-sm md:block ${
                        active
                          ? 'font-semibold text-foreground'
                          : completed
                            ? 'text-foreground-secondary'
                            : 'text-foreground-muted'
                      }`}
                    >
                      {step.label}
                    </span>
                  </button>

                  {index < steps.length - 1 && (
                    <div className='mx-3 h-px w-10 bg-border md:w-16' />
                  )}
                </div>
              );
            })}
          </div>
        )}
      </div>
    </header>
  );
};
```

- [ ] **Step 3: WizardClient에서 StepNavigation 제거, Header에 props 전달**

`src/components/pages/WizardClient.tsx`를 다음으로 교체한다:

```tsx
'use client';

import { useEffect } from 'react';
import { FileSelection } from '@/components/layout/FileSelection';
import { MetadataSettingsForm } from '@/components/layout/MetadataSettings';
import { ResultsViewer } from '@/components/layout/ResultsViewer';
import { useAppReducer } from '@/hooks/useAppReducer';
import type { MetadataSettings } from '@/types/metadata.types';

export function WizardClient() {
  const {
    state,
    setFiles,
    setSettings,
    goToStep,
    setZipProgress,
    setProcessing,
    resetAll,
    processFiles,
  } = useAppReducer();

  useEffect(() => {
    if (typeof window === 'undefined') return;
    if (state.files.length === 0) return;

    const handleBeforeUnload = (e: BeforeUnloadEvent) => {
      e.preventDefault();
    };
    window.addEventListener('beforeunload', handleBeforeUnload);
    return () => window.removeEventListener('beforeunload', handleBeforeUnload);
  }, [state.files.length]);

  const handleFileSelect = (files: File[]) => setFiles(files);

  const handleSettingsChange = (name: string, value: string | Date) => {
    setSettings((prev) => ({ ...prev, [name]: value }) as MetadataSettings);
  };

  const handleProcessFiles = async () => {
    const results = await processFiles();
    if (results.images.length > 0 || results.errors.length > 0) goToStep(3);
  };

  return (
    <section className='flex-1 overflow-auto'>
      <div className='mx-auto w-full max-w-6xl'>
        {state.activeStep === 1 && (
          <FileSelection
            activeStep={1}
            onFileSelect={handleFileSelect}
            sortedFiles={state.sortedFiles}
            goToStep={goToStep}
          />
        )}
        {state.activeStep === 2 && (
          <MetadataSettingsForm
            activeStep={2}
            settings={state.settings}
            onSettingsChange={handleSettingsChange}
            sortedFiles={state.sortedFiles}
            processing={state.processing}
            completed={state.completed}
            formRef={{ current: null }}
            goToStep={goToStep}
            onProcessFiles={handleProcessFiles}
          />
        )}
        {state.activeStep === 3 && (
          <ResultsViewer
            activeStep={3}
            resultRef={{ current: null }}
            resultImages={state.resultImages}
            errors={state.errors}
            processing={state.processing}
            zipProgress={state.zipProgress}
            setZipProgress={setZipProgress}
            setProcessing={setProcessing}
            goToStep={goToStep}
          />
        )}
      </div>
    </section>
  );
}
```
